<script>
const currentLang = "tr";

function toggleFavorite(plantId, btn) {
  const formData = new FormData();
  formData.append('plant_id', plantId);
  fetch('favorite_api.php', { method: 'POST', body: formData })
  .then(res => res.json())
  .then(data => {
    if (data.success && (data.action === 'added' || data.status === 'added')) btn.innerHTML = '❤️';
    else if (data.success && (data.action === 'removed' || data.status === 'removed')) btn.innerHTML = '🤍';
    else alert(currentLang === 'en' ? 'Please log in!' : 'Giriş yapmalısınız!');
  });
}

function generatePlantAIAdvice(temp, humidity, windSpeed, lang) {
  let notes = [];

  if (temp < 10) {
    notes.push(lang === 'en' 
      ? '❄️ <strong>COLD WEATHER WARNING:</strong> Move delicate plants indoors. Reduce watering.' 
      : '❄️ <strong>SOĞUK HAVA UYARISI:</strong> Hassas bitkilerinizi içeri alın. Sulamayı azaltın.');
  } else if (temp >= 10 && temp <= 25) {
    notes.push(lang === 'en' 
      ? '🌱 <strong>IDEAL GROWTH CONDITION:</strong> Weather is perfect for photosynthesis and plant growth.' 
      : '🌱 <strong>İDEAL BÜYÜME HAVA KOŞULU:</strong> Hava fotosentez ve bitki gelişimi için mükemmel.');
  } else {
    notes.push(lang === 'en' 
      ? '🔥 <strong>HIGH TEMPERATURE:</strong> Increased evaporation! Check soil moisture daily.' 
      : '🔥 <strong>YÜKSEK SICAKLIK UYARISI:</strong> Buharlaşma fazla. Toprak nemini her gün kontrol edin.');
  }

  if (humidity < 40) {
    notes.push(lang === 'en' 
      ? '💧 <strong>LOW HUMIDITY:</strong> Dry air. Spray water on tropical plant leaves.' 
      : '💧 <strong>DÜŞÜK NEM:</strong> Hava kuru. Tropikal bitki yapraklarına su püskürtün.');
  } else if (humidity > 80) {
    notes.push(lang === 'en' 
      ? '🛡️ <strong>HIGH HUMIDITY:</strong> Ensure ventilation to prevent fungal diseases.' 
      : '🛡️ <strong>YÜKSEK NEM:</strong> Mantar oluşumunu önlemek için alanı havalandırın.');
  }

  if (windSpeed > 25) {
    notes.push(lang === 'en' 
      ? '🌬️ <strong>STRONG WIND:</strong> Protect tall potted plants from tipping over.' 
      : '🌬️ <strong>Sert RÜZGAR:</strong> Yüksek saksılı bitkileri devrilmeye karşı koruyun.');
  }

  return notes.join('<br>');
}

function runAIWeatherEngine(coords) {
  const parts = coords.split(',');
  const lat = parts[0];
  const lon = parts[1];

  const statusElem = document.getElementById('aiStatus');
  const recEl = document.getElementById('aiRecommendation');

  if (statusElem) {
    statusElem.innerText = currentLang === 'en' ? '🔄 Processing AI Data...' : '🔄 AI Verileri İşliyor...';
  }

  fetch(`https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current=temperature_2m,relative_humidity_2m,wind_speed_10m`)
    .then(res => res.json())
    .then(data => {
      if (data && data.current) {
        const c = data.current;
        document.getElementById('weatherTemp').innerText = `${c.temperature_2m}°C`;
        document.getElementById('weatherHumidity').innerText = `%${c.relative_humidity_2m}`;
        document.getElementById('weatherWind').innerText = `${c.wind_speed_10m} km/h`;

        const aiAdvice = generatePlantAIAdvice(c.temperature_2m, c.relative_humidity_2m, c.wind_speed_10m, currentLang);
        
        if (recEl) recEl.innerHTML = aiAdvice;
        if (statusElem) {
          statusElem.innerText = currentLang === 'en' ? '● AI Analysis Completed' : '● AI Analizi Tamamlandı';
        }
      }
    })
    .catch(err => {
      console.error('Weather API Error:', err);
      if (recEl) recEl.innerHTML = currentLang === 'en' ? '❌ Unable to fetch climate data.' : '❌ İklim verisi çekilemedi.';
    });
}

document.addEventListener('DOMContentLoaded', function() {
  const citySelect = document.getElementById('citySelect');
  if (citySelect) {
    citySelect.addEventListener('change', function() {
      runAIWeatherEngine(this.value);
    });
    runAIWeatherEngine(citySelect.value);
  }
});
